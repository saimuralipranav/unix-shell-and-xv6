void print_prompt(const char *home_dir)
{
    char cwd[PATH_MAX];
    char hostname[HOST_NAME_MAX + 1];

    getcwd(cwd, sizeof(cwd));
    gethostname(hostname, sizeof(hostname));

    struct passwd *pw = getpwuid(getuid());

    char display_path[PATH_MAX];

    if (strcmp(cwd, home_dir) == 0) {
        strcpy(display_path, "~");
    } else if (strncmp(cwd, home_dir, strlen(home_dir)) == 0 &&
               cwd[strlen(home_dir)] == '/') {
        snprintf(display_path, sizeof(display_path),
                 "~%s", cwd + strlen(home_dir));
    } else {
        strcpy(display_path, cwd);
    }

    printf("<%s@%s:%s> ", pw->pw_name, hostname, display_path);
    fflush(stdout);
}

2. Parser

parser.c takes the tokens from the lexer and checks whether their sequence follows the grammar.

For example:

echo hello > out.txt

is valid:

But:

is invalid because > requires a TGT (WORD) after it.

Likewise:

is invalid because a line cannot start with OP_PIPE.

#include "lexer.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

static Token *new_token(TokenType type, const char *value)
{
    Token *token = malloc(sizeof(Token));
    if (token == NULL)
        return NULL;

    token->type = type;
    token->value = value ? strdup(value) : NULL;
    token->next = NULL;

    if (value && token->value == NULL) {
        free(token);
        return NULL;
    }

    return token;
}

static void append_token(Token **head, Token **tail, Token *token)
{
    if (*head == NULL)
        *head = token;
    else
        (*tail)->next = token;

    *tail = token;
}

Token *lex_line(const char *input)
{
    Token *head = NULL;
    Token *tail = NULL;
    size_t i = 0;

    while (input[i] != '\0') {

        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        /* Operators */
        if (input[i] == '|') {
            append_token(&head, &tail, new_token(TOKEN_PIPE, "|"));
            i++;
            continue;
        }

        if (input[i] == '&') {
            append_token(&head, &tail, new_token(TOKEN_AMP, "&"));
            i++;
            continue;
        }

        if (input[i] == ';') {
            append_token(&head, &tail, new_token(TOKEN_SEMI, ";"));
            i++;
            continue;
        }

        if (input[i] == '<') {
            append_token(&head, &tail, new_token(TOKEN_LT, "<"));
            i++;
            continue;
        }

        if (input[i] == '>') {
            if (input[i + 1] == '>') {
                append_token(&head, &tail, new_token(TOKEN_GTGT, ">>"));
                i += 2;
            } else {
                append_token(&head, &tail, new_token(TOKEN_GT, ">"));
                i++;
            }
            continue;
        }

        /* WORD */
        char word[1025];
        size_t j = 0;

        while (input[i] != '\0' &&
               !isspace((unsigned char)input[i]) &&
               input[i] != '|' &&
               input[i] != '&' &&
               input[i] != ';' &&
               input[i] != '<' &&
               input[i] != '>') {

            char c = input[i];

            /* Backslash outside quotes */
            if (c == '\\') {
                if (input[i + 1] == '\0') {
                    printf("cshell: invalid syntax\n");
                    free_tokens(head);
                    return NULL;
                }

                word[j++] = input[i + 1];
                i += 2;
                continue;
            }

            /* Single quote */
            if (c == '\'') {
                i++;

                while (input[i] != '\'' && input[i] != '\0')
                    word[j++] = input[i++];

                if (input[i] == '\0') {
                    printf("cshell: invalid syntax\n");
                    free_tokens(head);
                    return NULL;
                }

                i++;
                continue;
            }

            /* Double quote */
            if (c == '"') {
                i++;

                while (input[i] != '"' && input[i] != '\0') {
                    if (input[i] == '\\') {
                        if (input[i + 1] == '\0') {
                            printf("cshell: invalid syntax\n");
                            free_tokens(head);
                            return NULL;
                        }

                        if (input[i + 1] == '"' ||
                            input[i + 1] == '\\') {
                            word[j++] = input[i + 1];
                            i += 2;
                        } else {
                            word[j++] = input[i++];
                            word[j++] = input[i++];
                        }
                    } else {
                        word[j++] = input[i++];
                    }
                }

                if (input[i] == '\0') {
                    printf("cshell: invalid syntax\n");
                    free_tokens(head);
                    return NULL;
                }

                i++;
                continue;
            }

            word[j++] = c;
            i++;
        }

        word[j] = '\0';

        if (j > 0) {
            Token *token = new_token(TOKEN_WORD, word);

            if (token == NULL) {
                free_tokens(head);
                return NULL;
            }

            append_token(&head, &tail, token);
        }
    }

    return head;
}

void free_tokens(Token *tokens)
{
    while (tokens != NULL) {
        Token *next = tokens->next;
        free(tokens->value);
        free(tokens);
        tokens = next;
    }
}

4. src/parser.c

The grammar is right-linear, so you can implement it as a small state machine.

#include "parser.h"

static int is_word(Token *t)
{
    return t != NULL && t->type == TOKEN_WORD;
}

/*
 * ARG -> ε
 *      | WORD ARG
 *      | < TGT
 *      | > TGT
 *      | >> TGT
 *      | | CMD
 *      | ; CMD
 *      | & BG
 */
static int parse_arg(Token **current);

static int parse_cmd(Token **current)
{
    if (!is_word(*current))
        return 0;

    (*current)++;
    return parse_arg(current);
}

static int parse_tgt(Token **current)
{
    if (!is_word(*current))
        return 0;

    (*current)++;
    return parse_arg(current);
}

static int parse_bg(Token **current)
{
    if (*current == NULL)
        return 1;

    if (!is_word(*current))
        return 0;

    (*current)++;
    return parse_arg(current);
}

static int parse_arg(Token **current)
{
    if (*current == NULL)
        return 1;

    switch ((*current)->type) {

    case TOKEN_WORD:
        (*current)++;
        return parse_arg(current);

    case TOKEN_LT:
    case TOKEN_GT:
    case TOKEN_GTGT:
        (*current)++;
        return parse_tgt(current);

    case TOKEN_PIPE:
    case TOKEN_SEMI:
        (*current)++;
        return parse_cmd(current);

    case TOKEN_AMP:
        (*current)++;
        return parse_bg(current);
    }

    return 0;
}

int parse_line(Token *tokens)
{
    if (tokens == NULL)
        return 1;       /* empty line is valid */

    if (!is_word(tokens))
        return 0;       /* LINE must begin with WORD */

    tokens = tokens->next;

    return parse_arg(&tokens) && tokens == NULL;
}

Frecency

Keep a persistent record of directories hopped into, across shell sessions.

Every successful hop updates that directory’s standing.
Standing depends on both frequency and recency of visits.
When “name” doesn’t resolve directly, hop into the highest ranked directory whose path contains “name” as a substring.
If the top match no longer exists on disk, skip it and try the next best.

Scoring, decay, and storage are up to you, but must be deterministic.

Resource: same idea behind zoxide’s cd replacement: 
https://github.com/ajeetdsouza/zoxide/wiki/Algorithm
what is this

It basically means hop should remember directories you visit frequently, so later you can type a short name and jump there.

Example:

hop /home/pranav/projects
hop /home/pranav/Downloads
hop /home/pranav/projects
hop /home/pranav/projects

Your shell stores something like:

Directory                 Visits    Recent
/home/pranav/projects       3        high
/home/pranav/Downloads      1        medium

Later:

hop proj

If proj isn't a direct path, search your stored directories for paths containing "proj":

/home/pranav/projects
/home/pranav/old-projects

Then choose the highest-ranked one.

What does "frequency + recency" mean?

You get to choose the exact formula. For example:

score = frequency × 10 + recency_score

So a directory visited many times gets a high score, but recently visited directories also get boosted.

"Persistent across sessions"

This is the important part.

If you close your shell:

./shell.out
hop ~/projects
exit

and start it again tomorrow, the shell should still remember ~/projects.

So you need to save the data somewhere, e.g.:

~/.cshell_frecency

Conceptually:

path=/home/pranav/projects visits=5 last_visit=...
path=/home/pranav/Downloads visits=2 last_visit=...

You do not need to copy zoxide's algorithm. The assignment explicitly says the scoring, decay, and storage method are up to you; it only points you toward zoxide as inspiration.

For this project, I'd implement a simple deterministic frequency + timestamp score rather than trying to recreate zoxide.

1. Basic hop

Keep two paths in your shell state:

char home_dir[PATH_MAX];
char prev_dir[PATH_MAX];
int has_prev_dir;

Then:

int hop_to(const char *arg)
{
    char old_dir[PATH_MAX];
    char target[PATH_MAX];

    if (getcwd(old_dir, sizeof(old_dir)) == NULL)
        return -1;

    if (strcmp(arg, "~") == 0) {
        strcpy(target, home_dir);
    }
    else if (strcmp(arg, ".") == 0) {
        return 0;
    }
    else if (strcmp(arg, "..") == 0) {
        strcpy(target, "..");
    }
    else if (strcmp(arg, "-") == 0) {
        if (!has_prev_dir)
            return 0;

        strcpy(target, prev_dir);
    }
    else {
        strcpy(target, arg);
    }

    if (chdir(target) == 0) {
        strcpy(prev_dir, old_dir);
        has_prev_dir = 1;

        return 0;
    }

    return -1;
}
2. Handle multiple arguments

The requirement says arguments execute sequentially:

hop goatam/osnmp1 .. -

So:

for (int i = 1; i < argc; i++) {
    if (hop_to(argv[i]) != 0) {
        // try frecency here
        printf("hop: no such directory\n");
    }
}

This means after the first hop, the second argument operates on the new CWD.

3. Direct path before frecency

For a normal name:

if (chdir(arg) == 0) {
    // successful direct path
}
else {
    // search frecency
}

This automatically handles both:

hop foo
hop /tmp/foo

and allows paths outside the home directory, which the assignment explicitly requires.

4. Frecency

Keep a file such as:

~/.cshell_frecency

with entries like:

/path/to/project 5 1756123000
/path/to/downloads 2 1756122900

Meaning:

path    frequency    last_visit

When a direct chdir(name) fails:

Read the saved entries.
Keep entries whose path contains name.
Ignore directories that no longer exist.
Calculate a deterministic score from frequency + recency.
Pick the highest score.
chdir() there.
Update its frequency and timestamp.

For example:

score = frequency * 1000000 + last_visit;

This is deliberately simple and deterministic.

Syntax: locate filename+

Purpose: The locate command returns the pathnames of the files (or links) which would be executed in the current environment, had its arguments been given as commands in a strictly POSIX-conformant shell.

Requirements:

Core function:

void locate_command(const char *name)
{
    char path[PATH_MAX];
    char abs_path[PATH_MAX];
    int found = 0;

    /* 1. Check current directory */
    if (getcwd(path, sizeof(path)) != NULL) {
        snprintf(abs_path, sizeof(abs_path), "%s/%s", path, name);

        if (access(abs_path, X_OK) == 0) {
            printf("%s\n", abs_path);
            found = 1;
        }
    }

    /* 2. Search PATH */
    char *path_env = getenv("PATH");

    if (path_env != NULL) {
        char *copy = strdup(path_env);
        char *dir = strtok(copy, ":");

        while (dir != NULL) {
            snprintf(abs_path, sizeof(abs_path), "%s/%s", dir, name);

            if (access(abs_path, X_OK) == 0) {
                printf("%s\n", abs_path);
                found = 1;
            }

            dir = strtok(NULL, ":");
        }

        free(copy);
    }

    /* 3. Nothing found */
    if (!found)
        printf("locate: command not found (%s)\n", name);
}

char *resolve_command(const char *cmd)
{
    static char path[PATH_MAX];

    /* %name → PATH directly */
    if (cmd[0] == '%') {
        cmd++;
        return search_path(cmd, path);
    }

    /* Contains / → literal path */
    if (strchr(cmd, '/') != NULL) {
        if (is_executable(cmd))
            return (char *)cmd;
        return NULL;
    }

    /* First current directory */
    snprintf(path, sizeof(path), "./%s", cmd);

    if (is_executable(path))
        return path;

    /* Then PATH */
    return search_path(cmd, path);
}

int is_executable(const char *path)
{
    struct stat st;

    return stat(path, &st) == 0 &&
           S_ISREG(st.st_mode) &&
           access(path, X_OK) == 0;
}

Reuse the same logic you already implemented for locate:

char *search_path(const char *cmd, char *result)
{
    char *env = getenv("PATH");
    if (!env)
        return NULL;

    char *copy = strdup(env);
    char *dir = strtok(copy, ":");

        if (is_executable(result)) {
            free(copy);
            return result;

        dir = strtok(NULL, ":");

For C2, the important part is handling multiple < files as one continuous stdin stream.

The cleanest approach is to create a pipe, write all input files into the pipe, then connect the command's stdin to the pipe.

In the child process
int pipefd[2];

if (pipe(pipefd) == -1) {
    perror("pipe");
    exit(1);
}

pid_t pid = fork();

if (pid == 0) {
    // Child

    close(pipefd[1]);

    if (dup2(pipefd[0], STDIN_FILENO) == -1) {
        perror("dup2");
        exit(1);
    }

    close(pipefd[0]);

    execv(command_path, argv);

    perror("cshell");
    exit(1);
}
In the parent

Open each < file and write its contents into the pipe in order:

close(pipefd[0]);

for (int i = 0; i < input_file_count; i++) {

    int fd = open(input_files[i], O_RDONLY);

    if (fd == -1) {
        printf("cshell: no such file or directory\n");
        close(pipefd[1]);
        waitpid(pid, NULL, 0);
        return;
    }

    char buffer[4096];
    ssize_t n;

    while ((n = read(fd, buffer, sizeof(buffer))) > 0) {
        write(pipefd[1], buffer, n);
    }

    close(fd);
}

close(pipefd[1]);
waitpid(pid, NULL, 0);

So:

        ↓
open a.txt → pipe
open b.txt → pipe
        ↓
child stdin ← pipe
        ↓
cat sees: contents(a.txt) + contents(b.txt)

C3: Output Redirection [12]

Syntax: command > filename or command >> filename

Purpose: The output redirection operators allow a command’s output to be written to one or more files instead of the terminal.

Requirements

A command may have more than one output redirection (e.g. command > f1 >> f2 > f3). Every listed file must receive the complete output of the command.
Each redirection keeps its own mode independently: > truncates its file (or creates it with permissions 0644 if it doesn’t exist), >> appends to its file (or creates it with 0644 if it doesn’t exist).
When output redirection is present, the command’s output must not appear on the terminal, only in the listed files.
If any listed file cannot be opened for writing, the shell must print cshell: unable to create file for writing and not execute the command.
Input and output redirection must work together (e.g. command < input.txt > output.txt >> log.txt).

Example

<goatam@iiit:~> echo hi > out1.txt > out2.txt
<goatam@iiit:~> cat out1.txt
hi
<goatam@iiit:~> cat out2.txt
hi
<goatam@iiit:~> echo again >> out1.txt > out3.txt
<goatam@iiit:~> cat out1.txt
hi
again
<goatam@iiit:~> cat out3.txt
again

parent reads output
   ↓
